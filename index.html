    esp32_smart_feeding_rtc.ino (WiFi + MQTT) 
    #include <WiFi.h>
#include <PubSubClient.h>
#include <Wire.h>
#include "RTClib.h"

// ================= WIFI =================
const char* WIFI_SSID = "Samsung A17";
const char* WIFI_PASSWORD = "Fyreablaze";

// ================= MQTT =================
const char* MQTT_SERVER = "broker.emqx.io";
const int MQTT_PORT = 1883;

const char* TOPIC_RTC_DATA = "smartfeeding/rtc/data";
const char* TOPIC_RTC_LOG = "smartfeeding/rtc/log";
const char* TOPIC_PUMP_STATUS = "smartfeeding/pump/status";
const char* TOPIC_PUMP_COMMAND = "smartfeeding/pump/command";
const char* TOPIC_SCHEDULE = "smartfeeding/rtc/schedule";

// ================= PIN ESP32 =================
#define Pump1 19
#define Pump2 18
#define BT 5

RTC_DS3231 rtc;
WiFiClient espClient;
PubSubClient mqtt(espClient);

// ================= JADWAL PAKAN (4 KALI) =================
const int jadwalJam[]   = {6, 12, 15, 18};
const int jadwalMenit[] = {0, 0, 15, 0};
const int JUMLAH_JADWAL = 4;

bool jadwalSudahJalan[JUMLAH_JADWAL] = {false, false, false, false};
int hariTerakhir = -1, bulanTerakhir = -1, tahunTerakhir = -1;

// Status kontrol mandiri Pump 1 (Pakan) & Pump 2 (Air)
bool mqttPump1On = false;
bool mqttPump2On = false;
bool statusP1Sebelumnya = false;
bool statusP2Sebelumnya = false;

unsigned long waktuKirimTerakhir = 0;
const unsigned long INTERVAL_KIRIM = 1000;

void koneksiWiFi() {
  Serial.print("Menghubungkan ke WiFi: ");
  Serial.println(WIFI_SSID);
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi terhubung! IP: " + WiFi.localIP().toString());
}

void callback(char* topic, byte* payload, unsigned int length) {
  String pesan = "";
  for (unsigned int i = 0; i < length; i++) pesan += (char)payload[i];
  pesan.trim();
  pesan.toUpperCase();

  if (String(topic) == TOPIC_PUMP_COMMAND) {
    // Kontrol Pump 1: Dispenser Pakan (GPIO 19)
    if (pesan == "PUMP1 ON" || pesan == "PUMP1_ON" || pesan == "PUMP 1 ON") {
      mqttPump1On = true;
      Serial.println("Perintah: Pump 1 (Pakan) ON");
    } else if (pesan == "PUMP1 OFF" || pesan == "PUMP1_OFF" || pesan == "PUMP 1 OFF") {
      mqttPump1On = false;
      Serial.println("Perintah: Pump 1 (Pakan) OFF");
    }
    // Kontrol Pump 2: Pompa Air Minum (GPIO 18)
    else if (pesan == "PUMP2 ON" || pesan == "PUMP2_ON" || pesan == "PUMP 2 ON") {
      mqttPump2On = true;
      Serial.println("Perintah: Pump 2 (Air) ON");
    } else if (pesan == "PUMP2 OFF" || pesan == "PUMP2_OFF" || pesan == "PUMP 2 OFF") {
      mqttPump2On = false;
      Serial.println("Perintah: Pump 2 (Air) OFF");
    }
    // Kompatibilitas kontrol serentak
    else if (pesan == "PUMP ON" || pesan == "ON" || pesan == "ALL ON") {
      mqttPump1On = true; mqttPump2On = true;
    } else if (pesan == "PUMP OFF" || pesan == "OFF" || pesan == "ALL OFF") {
      mqttPump1On = false; mqttPump2On = false;
    }
  }
}

void koneksiMQTT() {
  while (!mqtt.connected()) {
    Serial.print("Menghubungkan ke MQTT...");
    String clientId = "SmartFeeding-" + String((uint32_t)ESP.getEfuseMac(), HEX);
    if (mqtt.connect(clientId.c_str())) {
      Serial.println("Terhubung!");
      mqtt.subscribe(TOPIC_PUMP_COMMAND);
      kirimStatusPompa();
    } else {
      Serial.print("Gagal, kode: ");
      Serial.println(mqtt.state());
      delay(3000);
    }
  }
}

void kirimStatusPompa() {
  char statusPayload[64];
  snprintf(statusPayload, sizeof(statusPayload), "P1:%s,P2:%s", mqttPump1On ? "ON" : "OFF", mqttPump2On ? "ON" : "OFF");
  mqtt.publish(TOPIC_PUMP_STATUS, statusPayload, true);
}

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22);

  pinMode(Pump1, OUTPUT);
  pinMode(Pump2, OUTPUT);
  pinMode(BT, INPUT_PULLUP);

  digitalWrite(Pump1, HIGH); // Relay OFF saat mulai (Aktif LOW)
  digitalWrite(Pump2, HIGH);

  if (!rtc.begin()) {
    Serial.println("RTC DS3231 tidak ditemukan!");
    while (true) delay(1000);
  }

  rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
  DateTime sekarang = rtc.now();
  hariTerakhir = sekarang.day();
  bulanTerakhir = sekarang.month();
  tahunTerakhir = sekarang.year();

  koneksiWiFi();
  mqtt.setServer(MQTT_SERVER, MQTT_PORT);
  mqtt.setCallback(callback);
  mqtt.setBufferSize(512);
}

void loop() {
  if (WiFi.status() != WL_CONNECTED) koneksiWiFi();
  if (!mqtt.connected()) koneksiMQTT();
  mqtt.loop();

  DateTime sekarang = rtc.now();

  // Reset jadwal jika hari berganti
  if (sekarang.day() != hariTerakhir || sekarang.month() != bulanTerakhir || sekarang.year() != tahunTerakhir) {
    hariTerakhir = sekarang.day(); bulanTerakhir = sekarang.month(); tahunTerakhir = sekarang.year();
    for (int i = 0; i < JUMLAH_JADWAL; i++) jadwalSudahJalan[i] = false;
  }

  // Periksa 4 jadwal pakan
  for (int i = 0; i < JUMLAH_JADWAL; i++) {
    if (sekarang.hour() == jadwalJam[i] && sekarang.minute() == jadwalMenit[i] && !jadwalSudahJalan[i]) {
      char pesanJadwal[100];
      snprintf(pesanJadwal, sizeof(pesanJadwal), "JADWAL_PAKAN|%02d:%02d", jadwalJam[i], jadwalMenit[i]);
      mqtt.publish(TOPIC_SCHEDULE, pesanJadwal);
      mqtt.publish(TOPIC_RTC_LOG, pesanJadwal);
      jadwalSudahJalan[i] = true;
    }
  }

  bool tombolDitekan = (digitalRead(BT) == LOW);
  bool p1Active = mqttPump1On || tombolDitekan;
  bool p2Active = mqttPump2On;

  digitalWrite(Pump1, p1Active ? LOW : HIGH);
  digitalWrite(Pump2, p2Active ? LOW : HIGH);

  // Kirim data RTC setiap 1 detik
  if (millis() - waktuKirimTerakhir >= INTERVAL_KIRIM) {
    waktuKirimTerakhir = millis();
    char tanggal[16], waktu[16], payload[320], logPesan[120];
    snprintf(tanggal, sizeof(tanggal), "%02d/%02d/%04d", sekarang.day(), sekarang.month(), sekarang.year());
    snprintf(waktu, sizeof(waktu), "%02d:%02d:%02d", sekarang.hour(), sekarang.minute(), sekarang.second());

    snprintf(payload, sizeof(payload),
      "{\"tanggal\":\"%s\",\"waktu\":\"%s\",\"pump1_status\":\"%s\",\"pump2_status\":\"%s\",\"pump_status\":\"%s\",\"mode\":\"mqtt\"}",
      tanggal, waktu, p1Active ? "ON" : "OFF", p2Active ? "ON" : "OFF", (p1Active || p2Active) ? "ON" : "OFF");
    mqtt.publish(TOPIC_RTC_DATA, payload);

    snprintf(logPesan, sizeof(logPesan), "Tanggal: %s | Waktu: %s | P1:%s P2:%s", tanggal, waktu, p1Active ? "ON" : "OFF", p2Active ? "ON" : "OFF");
    mqtt.publish(TOPIC_RTC_LOG, logPesan);

    if (p1Active != statusP1Sebelumnya || p2Active != statusP2Sebelumnya) {
      kirimStatusPompa();
      statusP1Sebelumnya = p1Active;
      statusP2Sebelumnya = p2Active;
    }
  }
}
